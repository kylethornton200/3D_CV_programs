cmake_minimum_required(VERSION 3.16)
project(lie_ekf_examples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Eigen3 CONFIG REQUIRED)

# ---- EKF examples (Eigen only) ----
add_executable(lie_ekf_examples lie_ekf_examples.cpp user_lie.hpp)
target_link_libraries(lie_ekf_examples PRIVATE Eigen3::Eigen)

# ---- Optional: tests of user_lie.hpp against Sophus (built only if Sophus is installed) ----
find_package(Sophus CONFIG QUIET)
if(Sophus_FOUND)
  add_executable(test_vs_sophus test_vs_sophus.cpp user_lie.hpp)
  target_link_libraries(test_vs_sophus PRIVATE Eigen3::Eigen Sophus::Sophus)
else()
  message(STATUS "Sophus not found - skipping test_vs_sophus")
endif()

# ---- Warnings / platform flags for every target above ----
foreach(tgt lie_ekf_examples test_vs_sophus)
  if(TARGET ${tgt})
    if(MSVC)
      target_compile_options(${tgt} PRIVATE /W4)
      target_compile_definitions(${tgt} PRIVATE _USE_MATH_DEFINES)  # makes M_PI available
    else()
      target_compile_options(${tgt} PRIVATE -Wall -Wextra)
    endif()
  endif()
endforeach()
